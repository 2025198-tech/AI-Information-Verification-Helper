// 디버깅 기능이 강화된 WebLLM 엔진 초기화 함수 async function initWebLLM() { const statusEl = document.getElementById("engineStatus"); const model = document.getElementById("modelSelect").value; const btn = document.getElementById("initEngineBtn"); const verifyBtn = document.getElementById("verifyBtn"); try { isEngineReady = false; verifyBtn.disabled = true; verifyBtn.className = "w-full bg-slate-300 text-slate-500 py-3 rounded-lg font-bold text-sm cursor-not-allowed transition"; statusEl.className = "text-sm font-semibold text-indigo-600"; statusEl.innerText = "🔍 WebGPU 환경 확인 중..."; btn.disabled = true; // 1. 브라우저의 WebGPU 지원 여부 사전 점검 if (!navigator.gpu) { throw new Error("WEBGPU_NOT_SUPPORTED"); } statusEl.innerText = "⏳ AI 모델 다운로드 및 메모리 로드 중..."; // 2. WebLLM 엔진 생성 engine = await window.webllm.CreateMLCEngine(model, { initProgressCallback: (progress) => { const pct = Math.round(progress.progress * 100); statusEl.innerText = `⏳ 모델 로딩 중: ${pct}% (${progress.text || ''})`; } }); isEngineReady = true; statusEl.className = "text-sm font-semibold text-emerald-600"; statusEl.innerText = "✅ AI 엔진 준비 완료!"; verifyBtn.disabled = false; verifyBtn.className = "w-full bg-emerald-600 hover:bg-emerald-700 text-white py-3 rounded-lg font-bold text-sm cursor-pointer transition shadow-md"; } catch (err) { isEngineReady = false; statusEl.className = "text-sm font-semibold text-red-600"; btn.disabled = false; // 구체적인 에러 원인 메시지 출력 if (err.message === "WEBGPU_NOT_SUPPORTED") { statusEl.innerText = "❌ 브라우저가 WebGPU를 지원하지 않거나 그래픽 가속이 꺼져 있습니다."; alert("현재 브라우저/PC에서 WebGPU를 지원하지 않습니다.\n크롬 설정에서 '그래픽 가속 사용'을 켜주시거나, 최신 크롬 브라우저로 접속해 주세요."); } else { statusEl.innerText = `❌ 엔진 로드 실패: ${err.message || '네트워크 또는 메모리 오류'}`; console.error("WebLLM 상세 에러 내역:", err); } } }
